## 设计备忘录

### Show 内部伪代码

```ts
function Show(props, context): HostNode[] {
  const adapter = getAdapter();
  const anchor = adapter.createComment("show");
  context.owner.elements.add(anchor);

  let result: HResult | null = null;

  const branch = (Component) => {
    const r = h(Component);
    r.owner!.parent = context.owner;
    context.owner.children.push(r.owner!);
    for (const node of r.nodes) {
      adapter.before(anchor, node);
    }
    return r;
  };

  if (toValue(props.value)) {
    result = branch(props.children[0]);
  } else if (props.children[1]) {
    result = branch(props.children[1]);
  }

  if (isUse(props.value)) {
    const derived = use(props.value, () => {
      if (result) disposeOwner(result.owner);
      result = null;
      if (toValue(props.value)) {
        result = branch(props.children[0]);
      } else if (props.children[1]) {
        result = branch(props.children[1]);
      }
    });
    const state = getSignalState(derived);
    if (state?.stop) context.owner.cleanups.push(state.stop);
  }

  return [anchor];
}
```

### Case 内部伪代码

与 Show 类似，但分支选择基于映射表查找：

```ts
const map = props.children[0];
const Component = map[String(toValue(props.value))] || props.children[1];
if (Component) result = branch(Component);
```

需要跟踪 `prevKey` 来避免同 key 重复切换。

### Each 条目响应式包裹

```ts
// 条目信号
const rawValue = array[i];
const itemSignal = isUse(rawValue) ? rawValue : definitionMode(rawValue);
// 传递给 h(ItemComponent, { item: itemSignal, index: i })
```

或者直接从数组值创建信号：

```ts
const itemSignal = definitionMode(rawValue);
// 当数组更新且 identity 匹配时，调用 itemSignal(newValue)
```
